import { CardGrid, Card } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="Logs You Can Assert On" icon="document">
    `LoggerTest` wires Serilog to your test framework's output and lets you capture log events inside a block and assert on them.
  </Card>
  <Card title="Auto-Mocking Containers" icon="puzzle">
    `AutoFakeTest` and `AutoSubstituteTest` resolve your system under test from a DryIoc container that fakes every dependency it doesn't know about.
  </Card>
  <Card title="Fixture Builders" icon="pencil">
    Write fluent builders with `ITestFixtureBuilder`, or let `[AutoFixture]` generate them from a constructor.
  </Card>
  <Card title="Generator Test Harness" icon="setting">
    `GeneratorTestContextBuilder` compiles sources, runs generators, analyzers, code fixes and refactorings, and snapshots the results with Verify.
  </Card>
</CardGrid>
